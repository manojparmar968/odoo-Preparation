# To Generate ssh-rsa key
    ssh-keygen -t rsa -b 4096 -C "email ID"
        > password
        > confirm password
    eval "$(ssh-agent -s)"
    ssh-add `/.ssh/id_rsa
        > Enter Password
    cat `/.ssh/id_rsa.pub
        > ssh-rsa key add into your git account to get access.

# To Generate ssh-rsa key for MacBook
    Step 1: Generate the SSH key
        ssh-keygen -t rsa -b 4096 -C "your-email@example.com"
    Step 2: Start the SSH agent
        eval "$(ssh-agent -s)"
    Step 3: Add the key to the macOS Keychain
        ssh-add --apple-use-keychain ~/.ssh/id_rsa
    Step 4: Configure SSH (recommended)
        mkdir -p ~/.ssh
        touch ~/.ssh/config
        nano ~/.ssh/config
            Add the following configuration:
                Host github.com
                    AddKeysToAgent yes
                    UseKeychain yes
                    IdentityFile ~/.ssh/id_rsa
            Note:- Save the file in Nano using Ctrl + O, press Enter, and then Ctrl + X.
            Set appropriate permissions:
                chmod 700 ~/.ssh
                chmod 600 ~/.ssh/config
                chmod 600 ~/.ssh/id_rsa
                chmod 644 ~/.ssh/id_rsa.pub
    Step 5: Display your public SSH key
        cat ~/.ssh/id_rsa.pub
    Step 6: Add the key to GitHub
        Open GitHub SSH settings --> Click New SSH key
    Step 7: Test your SSH connection
        ssh -T git@github.com
